#had to go back to rename R0ock and NA 

rolling_stone_clean <- rolling_stone |>
  mutate(
    `Album Genre` = recode(
      `Album Genre`,
      "Blues/Blues ROck" = "Blues/Blues Rock"
    ),
    `Album Genre` = replace_na(`Album Genre`, "Genre Unavailable")
  )
#using current dataset to create a new object to count alnums of each genre in 2003  

genre_2003 <- rolling_stone_clean |>
  filter(!is.na(`2003 Rank`)) |>
  count(`Album Genre`)
#using current dataset to create a new object to count alnums of each genre in 2012

genre_2012 <- rolling_stone_clean |>
  filter(!is.na(`2012 Rank`)) |>
  count(`Album Genre`)
#using current dataset to create a new object to count alnums of each genre in 2020  

genre_2020 <- rolling_stone_clean |>
  filter(!is.na(`2020 Rank`)) |>
  count(`Album Genre`)
#using full_join, going to take the rows that correspond and combine them together, thus, going to combine 2003,2012,2020 genres 

genre_comparison <- genre_2003 |>
  full_join(genre_2012, by = "Album Genre") |>
  full_join(genre_2020, by = "Album Genre")
#since genre_comparison is using random column names, going to change to legible ones 

genre_comparison_clean <- genre_comparison |> 
  rename(
    count_2003 = n.x, 
    count_2012 = n.y,
    count_2020 = n
  )
#going to be using | (or) to figure out when there is a genre missing at least once, to see when the genre appeared/left 

genre_comparison_clean |>
  filter(
    is.na(count_2003)|
    is.na(count_2012)|
    is.na(count_2020)
  )
# A tibble: 1 × 4
  `Album Genre` count_2003 count_2012 count_2020
  <chr>              <int>      <int>      <int>
1 Afrobeat              NA         NA          2
#as previously checked, the only NA is Afrobeat, so we can replace the NA to 0 since the previous years it just wasn't in the top 500

genre_comparison_clean <- genre_comparison_clean |>
  mutate(
    count_2003 = replace_na(count_2003, 0),
    count_2012 = replace_na(count_2012, 0),
    count_2020 = replace_na(count_2020, 0)
  )
# reordering data using average 

genre_comparison_clean <- genre_comparison_clean |>
  rowwise() |>
  mutate(
    mean_count = mean(c(count_2003, count_2012, count_2020))
  )|>
  arrange(mean_count)|>
  mutate(
    `Album Genre` = factor(`Album Genre`, `Album Genre`)
  )
#going to use pivot_longer() to change how the dot graph looks 

genre_plot <- genre_comparison_clean |>
  pivot_longer(
    cols = c(count_2003, count_2012, count_2020),
    names_to = "year",
    values_to = "count"
  )
